function starText(rating) {

    const full =
        Math.floor(rating);

    const half =
        rating - full >= .5;

    let result = "";

    for (let i = 0; i < full; i++) {
        result += "â˜…";
    }

    if (half &&
        result.length < 5) {

        result += "â˜…";
    }

    while (result.length < 5) {
        result += "â˜†";
    }

    return result;
}

function applyFilters() {

    const search =
        document
            .getElementById("searchInput")
            .value
            .toLowerCase()
            .trim();


    const category =
        document
            .getElementById("categoryFilter")
            .value;


    const price =
        document
            .getElementById("priceFilter")
            .value;


    const minRating =
        Number(
            document
                .getElementById("ratingFilter")
                .value
        );


    const sort =
        document
            .getElementById("sortFilter")
            .value;


    let visible =
        PRODUCTS.filter(
            function(product) {


                const matchSearch =
                    !search ||
                    product.name
                        .toLowerCase()
                        .includes(search) ||
                    product.category
                        .toLowerCase()
                        .includes(search) ||
                    product.description
                        .toLowerCase()
                        .includes(search);


                const matchCategory =
                    category === "ALL" ||
                    product.category === category;


                let matchPrice = true;


                if (price !== "ALL") {

                    const parts =
                        price.split("-");

                    const min =
                        Number(parts[0]);

                    const max =
                        Number(parts[1]);

                    matchPrice =
                        product.price >= min &&
                        product.price <= max;
                }


                const matchRating =
                    product.rating >= minRating;


                return (
                    matchSearch &&
                    matchCategory &&
                    matchPrice &&
                    matchRating
                );

            }
        );


    if (sort === "price-low") {

        visible.sort(
            (a,b) =>
                a.price - b.price
        );

    } else if (sort === "price-high") {

        visible.sort(
            (a,b) =>
                b.price - a.price
        );

    } else if (sort === "rating") {

        visible.sort(
            (a,b) =>
                b.rating - a.rating
        );

    } else if (sort === "name") {

        visible.sort(
            (a,b) =>
                a.name.localeCompare(
                    b.name
                )
        );

    }


    cards.forEach(function(card) {

        const id =
            Number(card.dataset.id);

        const show =
            visible.some(
                function(product) {
                    return product.id === id;
                }
            );

        card.style.display =
            show ? "block" : "none";

    });


    document
        .getElementById("productCount")
        .innerText =
            visible.length +
            " product(s) found";

}

function addToCart(
    event,
    id,
    name,
    image,
    price,
    stock
) {

    event.stopPropagation();

    if (stock <= 0) {

        showToast(
            "Product is out of stock"
        );

        return;

    }


    const cart =
        getCart();


    const existing =
        cart.find(
            function(item) {
                return Number(item.id) === id;
            }
        );


    if (existing) {

        existing.quantity =
            Number(existing.quantity || 1) + 1;

    } else {

        cart.push({

            id: id,

            name: name,

            image: image,

            price: price,

            quantity: 1

        });

    }


    saveCart(cart);

    showToast(
        "âœ“ Added to cart"
    );

}

    event,
    id,
    name,
    image,
    price,
    stock
) {

    event.stopPropagation();

    if (stock <= 0) {

        showToast(

        );

        return;

    }


    const cart = [

        {
            id: id,

            name: name,

            image: image,

            price: price,

            quantity: 1

        }

    ];


    saveCart(cart);

    window.location.href =
        "cart.html";

}

function processAIQuery(query) {

    const original =
        String(query || "");

    let q =
        original
            .toLowerCase()
            .replace(/â‚¹/g, " rs ")
            .replace(/,/g, "")
            .trim();


    /*
     * ==========================================
     * TAMIL + ENGLISH NORMALIZATION
     * ==========================================
     */

    const replacements = [

        [/\bkulla\b/g, " under "],
        [/\bkulllaa\b/g, " under "],
        [/\bukkulla\b/g, " under "],
        [/\bvaraikkum\b/g, " under "],
        [/\bvaraikum\b/g, " under "],
        [/\bvenum\b/g, " want "],
        [/\bvenunga\b/g, " want "],
        [/\bvennu\b/g, " want "],
        [/\benakku\b/g, " "],
        [/\bplease\b/g, " "],
        [/\bcheap\b/g, " low price "],
        [/\baffordable\b/g, " low price "],
        [/\bgood\b/g, " best "],
        [/\bbest ah\b/g, " best "],
        [/\bmathiri\b/g, " "],
        [/\bproductu\b/g, " product "]

    ];


    replacements.forEach(
        function(pair) {

            q = q.replace(
                pair[0],
                pair[1]
            );

        }
    );


    let result =
        PRODUCTS.slice();


    /*
     * ==========================================
     * PRICE / BUDGET
     * ==========================================
     */

    let maxPrice = null;


    const kMatch =
        q.match(
            /(?:under|below|within|upto|up to|less than)\s*(?:rs\s*)?(\d+(?:\.\d+)?)\s*k\b/i
        );


    if(kMatch) {

        maxPrice =
            Number(kMatch[1]) * 1000;

    }


    const lakhMatch =
        q.match(
            /(?:under|below|within|upto|up to)\s*(?:rs\s*)?(\d+(?:\.\d+)?)\s*lakh/i
        );


    if(lakhMatch) {

        maxPrice =
            Number(lakhMatch[1]) * 100000;

    }


    const priceMatch =
        q.match(
            /(?:under|below|within|upto|up to|less than|budget)\s*(?:rs\s*)?(\d+(?:\.\d+)?)/i
        );


    if(
        !maxPrice &&
        priceMatch
    ) {

        maxPrice =
            Number(priceMatch[1]);

    }


    if(maxPrice !== null &&
       !Number.isNaN(maxPrice)) {

        result =
            result.filter(
                function(product) {

                    return product.price <= maxPrice;

                }
            );

    }


    /*
     * ==========================================
     * CATEGORY / PRODUCT INTENT
     * ==========================================
     */

    const intentGroups = {

        laptop: [
            "laptop",
            "notebook",
            "computer"
        ],

        mobile: [
            "mobile",
            "phone",
            "smartphone",
            "iphone",
            "android"
        ],

        headphone: [
            "headphone",
            "headphones",
            "earphone",
            "earbuds",
            "earphone"
        ],

        keyboard: [
            "keyboard"
        ],

        mouse: [
            "mouse"
        ],

        monitor: [
            "monitor",
            "screen"
        ],

        watch: [
            "watch",
            "smartwatch"
        ],

        beauty: [
            "beauty",
            "skincare",
            "skin",
            "cosmetic",
            "makeup",
            "shampoo",
            "moisturizer",
            "conditioner",
            "perfume"
        ],

        fashion: [
            "fashion",
            "dress",
            "clothes",
            "shirt",
            "shoe"
        ],

        electronics: [
            "electronics",
            "electronic",
            "gadget"
        ],

        home: [
            "home",
            "house",
            "kitchen"
        ]

    };


    let intent =
        null;


    for(
        const key in intentGroups

        if(
            intentGroups[key].some(
                function(word) {

                }
            )
        ) {

            intent = key;
            break;

        }

    }


    if(intent) {

        const intentResult =
            result.filter(
                function(product) {

                    const text =
                        (
                            product.name +
                            " " +
                            product.category +
                            " " +
                            product.description
                        )
                        .toLowerCase();

                    return intentGroups[intent]
                        .some(
                            function(word) {

                                return text.includes(word);

                            }
                        );

                }
            );


        if(intentResult.length) {

            result =

        }

    }


    /*
     * ==========================================
     * ACTUAL DATABASE CATEGORY
     * ==========================================
     */

    const matchedCategory =
        categories.find(

                );

        );

    /*
     * ==========================================

    ];

            function(a,b) {

    if(

    }


    if(

        result.sort(
            function(a,b) {

                return b.price -
                    a.price;

            }
        );

    }


    /*
     * ==========================================
     * GENERIC KEYWORD MATCHING
     * ==========================================
     */

    const ignoredWords = [

        "under",
        "below",
        "within",
        "upto",
        "up",
        "to",
        "less",
        "than",
        "budget",
        "best",
        "top",
        "rated",
        "cheap",
        "cheapest",
        "expensive",
        "premium",
        "product",

        "show",

        "find",
        "want",
        "need",
        "please",
        "good",
        "price",
        "colour",
        "color"

    ];


    const words =
        q
            .replace(
                /[^a-z0-9 ]/g,
                " "
            )
            .split(/\s+/)
            .filter(
                function(word) {

                    return (
                        word.length >= 3 &&
                        !ignoredWords.includes(word) &&
                        !/^\d+$/.test(word)
                    );

                }
            );


    if(words.length) {

        const keywordResult =
            result.filter(
                function(product) {

                    const text =
                        (
                            product.name +
                            " " +
                            product.category +
                            " " +
                            product.description
                        )
                        .toLowerCase();


                    return words.some(
                        function(word) {

                            return text.includes(word);

                        }
                    );

                }
            );


        if(keywordResult.length) {

            result =
                keywordResult;

        }

    }


    /*
     * ==========================================
     * TOP RESULTS
     * ==========================================
     */

    result =
        result.slice(0,6);


    if(!result.length) {

        return `
            Sorry ðŸ˜”
            I couldn't find an exact match.

            <br><br>

            Try:
            <br>
            <b>50000 kulla best laptop venum</b>
            <br>
            <b>enakku phone 20000 kulla venum</b>
            <br>
            <b>cheap beauty products</b>
            <br>
            <b>best rated electronics</b>
        `;

    }


    const intro =
        maxPrice
        ? `I understood your budget as â‚¹${maxPrice.toLocaleString("en-IN")}.`
        : `I understood what you're looking for.`;

    const resultText =
        result
            .map(
                function(product,index) {

                    return `
                        <div
                            class="ai-result-product"
                            onclick="focusProduct(${product.id})">

            .join("");


    /*
     * Colour explanation
     */

    const colorText =

    return `

    `;
